Private Function Klasor() As Outlook.MAPIFolder
    Dim ana As Outlook.MAPIFolder, alt As Outlook.MAPIFolder
    Set ana = Application.Session.GetDefaultFolder(olFolderTasks)
    For Each alt In ana.Folders
        If StrComp(alt.Name, "Takip Kolay", vbTextCompare) = 0 Then
            Set Klasor = alt
            Exit Function
        End If
    Next
End Function

Private Function Alan(ByVal t As Outlook.TaskItem, ByVal ad As String) As String
    On Error GoTo Bitti
    Dim p As Outlook.UserProperty
    Set p = t.UserProperties.Find(ad)
    If Not p Is Nothing Then
        If Not IsNull(p.Value) Then Alan = CStr(p.Value)
    End If
Bitti:
End Function

Private Function Durum(ByVal t As Outlook.TaskItem) As String
    If t.Status = olTaskComplete Then
        Durum = "DONE"
    Else
        Durum = UCase$(Trim$(Alan(t, ALAN_DURUM)))
        If Len(Durum) = 0 Then Durum = "TODO"
    End If
End Function

Private Function DurumAdi(ByVal kod As String) As String
    Select Case kod
        Case "WAIT": DurumAdi = "Yanit bekliyor"
        Case "TODO": DurumAdi = "Yapacagim"
        Case "REVIEW": DurumAdi = "Incelenecek"
        Case "DONE": DurumAdi = "Tamamlandi"
        Case Else: DurumAdi = "Diger"
    End Select
End Function

Private Function FiltreUygun(ByVal kod As String) As Boolean
    If pencereTamamlanan Then
        ' Tamamlananlar yalnizca ayri pencerede gorunur.
        FiltreUygun = (kod = "DONE")
        Exit Function
    End If
    If kod = "DONE" Then Exit Function
    Select Case cmbFiltre.ListIndex
        Case 1: FiltreUygun = (kod = "TODO")
        Case 2: FiltreUygun = (kod = "WAIT")
        Case 3: FiltreUygun = (kod = "REVIEW")
        Case Else: FiltreUygun = True
    End Select
End Function

Private Sub Yukle(Optional ByVal odakID As String = "")
    On Error GoTo Hata
    Dim nesne As Object, t As Outlook.TaskItem, ogeler As Outlook.Items
    Dim konu As String, kisi As String, kod As String, vade As String
    Dim gun As Date, no As Long, i As Long, baslama As Date

    If Not arayuzHazir Then Exit Sub
    kayitSayisi = 0: tarananSayisi = 0: hataliKayit = 0
    okumaNotu = ""
    Erase kayitlar: Erase sira
    For i = 0 To 4: sayac(i) = 0: Next i
    ReDim kayitlar(1 To MAX_GOREV, 1 To 6)
    If takipKlasoru Is Nothing Then Set takipKlasoru = Klasor()
    If takipKlasoru Is Nothing Then
        KartlariYaz
        ListeyiGoster
        lblBilgi.Caption = "Takip klasoru bulunamadi. Once test e-postasini takibe alin."
        Exit Sub
    End If

    Set ogeler = takipKlasoru.Items
    ' En yeni kayitlari once oku. Sort sadece koleksiyonun sirasini etkiler.
    On Error Resume Next
    ogeler.Sort "[CreationTime]", True
    Err.Clear
    On Error GoTo Hata
    baslama = Now
    ' Burada e-posta govdesi veya konusma GORUNTULENMEZ; sadece gorev bilgisi.
    For Each nesne In ogeler
        If tarananSayisi >= MAX_GOREV Then
            okumaNotu = "Liste siniri: En yeni 600 oge okundu. Daha eskiler eksik olabilir."
            Exit For
        End If
        If DateDiff("s", baslama, Now) >= MAX_OKUMA_SANIYE Then
            okumaNotu = "Hiz icin okuma 7 saniyede durdu. Liste kismi olabilir."
            Exit For
        End If
        tarananSayisi = tarananSayisi + 1
        If TypeOf nesne Is Outlook.TaskItem Then
            Set t = nesne
            On Error GoTo KayitHatasi
            kod = Durum(t)
            konu = Alan(t, ALAN_KONU)
            If Len(konu) = 0 Then konu = t.Subject
            If Left$(konu, 8) = "[Takip] " Then konu = Mid$(konu, 9)
            kisi = Alan(t, ALAN_KISI)
            vade = "Tarihsiz"
            gun = 0
            If Year(t.DueDate) >= 2020 And Year(t.DueDate) <= 2100 Then
                gun = DateValue(t.DueDate)
                vade = Format$(gun, "dd.mm.yyyy")
                If kod <> "DONE" Then
                    If gun < Date Then vade = "Gecikti " & vade
                    If gun = Date Then vade = "Bugun " & vade
                End If
            End If
            kayitSayisi = kayitSayisi + 1
            no = kayitSayisi
            kayitlar(no, 1) = kod
            kayitlar(no, 2) = vade
            kayitlar(no, 3) = Left$(konu, 250)
            kayitlar(no, 4) = Left$(kisi, 150)
            kayitlar(no, 5) = t.EntryID
            If gun > 0 Then
                kayitlar(no, 6) = CDbl(gun)
            Else
                kayitlar(no, 6) = 999999#
            End If
            If kod = "DONE" Then kayitlar(no, 6) = CDbl(kayitlar(no, 6)) + 2000000#
            Select Case kod
                Case "TODO": sayac(1) = sayac(1) + 1
                Case "WAIT": sayac(2) = sayac(2) + 1
                Case "REVIEW": sayac(3) = sayac(3) + 1
                Case "DONE": sayac(4) = sayac(4) + 1
            End Select
            GoTo Devam
KayitHatasi:
            hataliKayit = hataliKayit + 1
            Err.Clear
            Resume Devam
Devam:
            On Error GoTo Hata
        End If
    Next nesne
    sayac(0) = kayitSayisi - sayac(4)

    If kayitSayisi > 0 Then
        ReDim sira(1 To kayitSayisi)
        For i = 1 To kayitSayisi: sira(i) = i: Next i
        Sirala
    End If
    KartlariYaz
    ListeyiGoster odakID
    If lstIsler.ListIndex >= 0 Then lstIsler_Click
    Exit Sub
Hata:
    lblBilgi.Caption = "Gorev listesi okunamadi. Hata " & Err.Number & ": " & Err.Description
End Sub

Private Sub Sirala()
    Dim i As Long, j As Long, tut As Long
    ' En fazla 600 kayıt; Outlook'a tekrar sorgu göndermeden bellek içinde sıralama.
    For i = 2 To kayitSayisi
        tut = sira(i)
        j = i - 1
        Do While j >= 1
            If CDbl(kayitlar(sira(j), 6)) <= CDbl(kayitlar(tut, 6)) Then Exit Do
            sira(j + 1) = sira(j)
            j = j - 1
        Loop
        sira(j + 1) = tut
    Next i
End Sub

Private Sub KartlariYaz()
    kartTum.Caption = "AÇIK İŞLER" & vbCrLf & CStr(sayac(0))
    kartYap.Caption = "YAPACAKLARIM" & vbCrLf & CStr(sayac(1))
    kartYanit.Caption = "YANIT BEKLEYEN" & vbCrLf & CStr(sayac(2))
    kartIncele.Caption = "İNCELENECEK" & vbCrLf & CStr(sayac(3))
    kartBiten.Caption = "TAMAMLANAN" & vbCrLf & CStr(sayac(4))
End Sub

Private Sub ListeyiGoster(Optional ByVal odakID As String = "")
    On Error GoTo Hata
    Dim i As Long, k As Long, ix As Long, bulunan As Long
    Dim q As String, metin As String, secimID As String
    If Not arayuzHazir Or listeGuncelleniyor Then Exit Sub
    listeGuncelleniyor = True
    secimID = odakID
    If Len(secimID) = 0 And lstIsler.ListIndex >= 0 Then
        secimID = CStr(lstIsler.List(lstIsler.ListIndex, 3))
    End If
    lstIsler.Clear
    q = Trim$(txtAra.Text)
    For i = 1 To kayitSayisi
        k = sira(i)
        If FiltreUygun(CStr(kayitlar(k, 1))) Then
            metin = CStr(kayitlar(k, 3)) & " " & CStr(kayitlar(k, 4)) & " " & _
                    DurumAdi(CStr(kayitlar(k, 1))) & " " & CStr(kayitlar(k, 2))
            If Len(q) = 0 Or InStr(1, metin, q, vbTextCompare) > 0 Then
                lstIsler.AddItem DurumAdi(CStr(kayitlar(k, 1)))
                ix = lstIsler.ListCount - 1
                lstIsler.List(ix, 1) = CStr(kayitlar(k, 2))
                lstIsler.List(ix, 2) = CStr(kayitlar(k, 3)) & _
                    IIf(Len(CStr(kayitlar(k, 4))) > 0, " | " & CStr(kayitlar(k, 4)), "")
                lstIsler.List(ix, 3) = CStr(kayitlar(k, 5))
                bulunan = bulunan + 1
                If Len(secimID) > 0 Then
                    If secimID = CStr(kayitlar(k, 5)) Then lstIsler.ListIndex = ix
                End If
            End If
        End If
    Next i
    If pencereTamamlanan Then
        lblBilgi.Caption = CStr(bulunan) & " tamamlanan kayit listeleniyor. " & _
                           CStr(kayitSayisi) & " takip kaydi okundu."
    Else
        lblBilgi.Caption = CStr(bulunan) & " acik is listeleniyor / " & _
                           CStr(sayac(0)) & " acik is. Arama bu listededir."
    End If
    If Len(okumaNotu) > 0 Then lblBilgi.Caption = lblBilgi.Caption & " " & okumaNotu
    If hataliKayit > 0 Then lblBilgi.Caption = lblBilgi.Caption & _
                                     " Okunamayan gorev: " & CStr(hataliKayit)
    If lstIsler.ListCount = 0 Then
        txtYazisma.Text = "Bu aramada veya kategoride is bulunamadi."
    ElseIf lstIsler.ListIndex = -1 Then
        txtYazisma.Text = "Soldan bir is secin; Outlook yazismasi burada gosterilecektir."
    End If
    listeGuncelleniyor = False
    Exit Sub
Hata:
    listeGuncelleniyor = False
    lblBilgi.Caption = "Liste hazirlanamadi. Hata " & Err.Number & ": " & Err.Description
End Sub

Private Sub kartBiten_Click()
    On Error GoTo Hata
    If pencereTamamlanan Then Exit Sub
    ' Formun ikinci kopyasi: yeni .frm/.frx veya yeni makro gerektirmez.
    ' Ayni anda tek bir "Tamamlananlar" penceresi acik tutulur.
    If Not tamamlananPenceresi Is Nothing Then
        On Error Resume Next
        If tamamlananPenceresi.Visible Then
            ' Pencere zaten acik, ikinci kopya olusturma.
            Exit Sub
        End If
        Set tamamlananPenceresi = Nothing
        On Error GoTo Hata
    End If
    Set tamamlananPenceresi = New frmRadarIsMasam
    tamamlananPenceresi.TamamlananPenceresiniHazirla
    tamamlananPenceresi.Show vbModeless
    Exit Sub
Hata:
    MsgBox "Tamamlananlar penceresi acilamadi. Hata " & Err.Number & _
           ": " & Err.Description, vbExclamation, "Is ve Yanit Radari"
End Sub

Private Function SeciliGorev() As Outlook.TaskItem
    On Error GoTo Bitti
    Dim id As String, obj As Object
    If lstIsler.ListIndex < 0 Then Exit Function
    If takipKlasoru Is Nothing Then Exit Function
    id = CStr(lstIsler.List(lstIsler.ListIndex, 3))
    If Len(id) = 0 Then Exit Function
    Set obj = Application.Session.GetItemFromID(id, takipKlasoru.StoreID)
    If TypeOf obj Is Outlook.TaskItem Then Set SeciliGorev = obj
Bitti:
End Function

Private Function KaynakMail(ByVal t As Outlook.TaskItem) As Outlook.MailItem
    On Error GoTo Bitti
    Dim kimlik As String, magaza As String, oge As Object
    kimlik = Alan(t, ALAN_KAYNAK)
    magaza = Alan(t, ALAN_STORE)
    If Len(kimlik) = 0 Then Exit Function
    If Len(magaza) > 0 Then
        Set oge = Application.Session.GetItemFromID(kimlik, magaza)
    Else
        Set oge = Application.Session.GetItemFromID(kimlik)
    End If
    If Not oge Is Nothing Then
        If TypeOf oge Is Outlook.MailItem Then Set KaynakMail = oge
    End If
Bitti:
End Function

Private Function MesajTarihi(ByVal m As Outlook.MailItem) As Date
    On Error Resume Next
    MesajTarihi = m.SentOn
    If Year(MesajTarihi) < 1990 Then MesajTarihi = m.ReceivedTime
    On Error GoTo 0
End Function

Private Function MesajAnahtar(ByVal m As Outlook.MailItem) As String
    On Error Resume Next
    MesajAnahtar = m.EntryID & "|" & m.Parent.StoreID
    If Len(MesajAnahtar) < 2 Then MesajAnahtar = m.Subject
    On Error GoTo 0
End Function

Private Sub KonusmaTopla(ByVal ogeler As Object, ByVal konusma As Outlook.Conversation, _
                        ByVal koleksiyon As Collection, ByVal gorulen As Object, _
                        ByVal seviye As Long, ByVal baslama As Date)
    On Error GoTo Bitti
    Dim oge As Object, alt As Object, key As String
    If ogeler Is Nothing Or seviye > 12 Then Exit Sub
    If DateDiff("s", baslama, Now) >= MAX_KONUSMA_SANIYE Then Exit Sub
    For Each oge In ogeler
        If koleksiyon.Count >= MAX_KONUSMA Then Exit For
        If DateDiff("s", baslama, Now) >= MAX_KONUSMA_SANIYE Then Exit For
        If TypeOf oge Is Outlook.MailItem Then
            key = MesajAnahtar(oge)
            If Not gorulen.Exists(key) Then
                gorulen.Add key, True
                koleksiyon.Add oge
            End If
        End If
        Set alt = Nothing
        On Error Resume Next
        Set alt = konusma.GetChildren(oge)
        Err.Clear
        On Error GoTo Bitti
        If Not alt Is Nothing Then KonusmaTopla alt, konusma, koleksiyon, gorulen, seviye + 1, baslama
    Next oge
Bitti:
End Sub

Private Function YazismaOku(ByVal kaynak As Outlook.MailItem) As String
    On Error GoTo Hata
    Dim konusma As Outlook.Conversation, kokler As Object
    Dim mailler As Collection, gorulen As Object
    Dim dizi() As Outlook.MailItem, tmp As Outlook.MailItem
    Dim i As Long, j As Long, adet As Long, ilk As Long
    Dim m As Outlook.MailItem, metin As String, satir As String
    Dim kimden As String, kime As String, konu As String, baslama As Date
    baslama = Now
    Set mailler = New Collection
    Set gorulen = CreateObject("Scripting.Dictionary")
    On Error Resume Next
    Set konusma = kaynak.GetConversation
    If Not konusma Is Nothing Then Set kokler = konusma.GetRootItems
    Err.Clear
    On Error GoTo Hata
    If Not kokler Is Nothing Then KonusmaTopla kokler, konusma, mailler, gorulen, 0, baslama
    If Not gorulen.Exists(MesajAnahtar(kaynak)) Then mailler.Add kaynak
    adet = mailler.Count
    ReDim dizi(1 To adet)
    For i = 1 To adet
        Set dizi(i) = mailler(i)
    Next i
    ' Yalnizca Outlook'un bulabildigi konusma kayitlari tarihe gore dizilir.
    For i = 2 To adet
        Set tmp = dizi(i)
        j = i - 1
        Do While j >= 1
            If MesajTarihi(dizi(j)) <= MesajTarihi(tmp) Then Exit Do
            Set dizi(j + 1) = dizi(j)
            j = j - 1
        Loop
        Set dizi(j + 1) = tmp
    Next i
    YazismaOku = "YAZISMA GECMISI (Outlook'un bulabildigi)" & vbCrLf & _
                  "Bulunan mesaj: " & CStr(adet) & vbCrLf & String$(58, "-") & vbCrLf
    If adet >= MAX_KONUSMA Then
        YazismaOku = YazismaOku & "Konusma ust sinirina ulasildi; tum yanitlar gorunmeyebilir." & vbCrLf
    End If
    If DateDiff("s", baslama, Now) >= MAX_KONUSMA_SANIYE Then
        YazismaOku = YazismaOku & "Hiz icin yazisma taramasi sinirlandi." & vbCrLf
    End If
    ilk = 1
    If adet > MAX_GOSTER Then
        YazismaOku = YazismaOku & "En son " & MAX_GOSTER & " mesaj gosteriliyor." & vbCrLf
        ilk = adet - MAX_GOSTER + 1
    End If
    YazismaOku = YazismaOku & vbCrLf
    For i = ilk To adet
        Set m = dizi(i)
        kimden = "Bilinmiyor": kime = "Bilinmiyor": konu = ""
        metin = "[Mesaj icerigi okunamadi]"
        ' Tek bir kilitli/erisim hatali posta diger mesajlari engellemesin.
        On Error Resume Next
        kimden = m.SenderName
        kime = m.To
        konu = m.Subject
        metin = m.Body
        Err.Clear
        On Error GoTo Hata
        If Len(Trim$(metin)) = 0 Then metin = "[Mesaj metni bos veya okunamadi]"
        If Len(metin) > MAX_MAIL_METNI Then
            metin = Left$(metin, MAX_MAIL_METNI) & vbCrLf & "[Devami kisaltildi]"
        End If
        satir = "MESAJ " & CStr(i) & " / " & CStr(adet) & vbCrLf & _
                Format$(MesajTarihi(m), "dd.mm.yyyy HH:nn") & _
                "  |  KIMDEN: " & Left$(kimden, 160) & vbCrLf & _
                "KIME: " & Left$(kime, 180) & vbCrLf & _
                "KONU: " & Left$(konu, 240) & vbCrLf & vbCrLf & _
                metin & vbCrLf & vbCrLf & String$(58, "-") & vbCrLf
        YazismaOku = YazismaOku & satir
    Next i
    Exit Function
Hata:
    YazismaOku = "Yazisma listesi okunamadi. Hata " & Err.Number & _
                 ". E-postayi Ac dugmesiyle asıl mesaji acmayi deneyin."
End Function

Private Sub OutlookAra(ByVal t As Outlook.TaskItem)
    On Error GoTo Hata
    Dim ex As Outlook.Explorer, konu As String
    konu = Alan(t, ALAN_KONU)
    If Len(konu) = 0 Then konu = t.Subject
    If Left$(konu, 8) = "[Takip] " Then konu = Mid$(konu, 9)
    konu = Replace(Replace(Replace(konu, vbCr, " "), vbLf, " "), """", "")
    konu = Left$(Trim$(konu), 90)
    If Len(konu) = 0 Then Exit Sub
    Set ex = Application.ActiveExplorer
    If ex Is Nothing Then
        Set ex = Application.Session.GetDefaultFolder(olFolderInbox).GetExplorer
        ex.Display
    Else
        Set ex.CurrentFolder = Application.Session.GetDefaultFolder(olFolderInbox)
    End If
    ex.Search "subject:""" & konu & """", olSearchScopeAllOutlookItems
    Exit Sub
Hata:
    MsgBox "Outlook aramasi acilamadi. Konuyu arama kutusuna yazabilirsiniz." & _
           vbCrLf & "Hata " & Err.Number & ": " & Err.Description, vbExclamation
End Sub

Private Sub HatirlatmaYanitTaslagi(ByVal tumune As Boolean)
    On Error GoTo Hata
    Dim t As Outlook.TaskItem, kaynak As Outlook.MailItem
    Dim taslak As Outlook.MailItem, govde As String
    Dim p As Long, son As Long, etiket As String
    Set t = SeciliGorev()
    If t Is Nothing Then
        MsgBox "Once soldan bir is secin.", vbInformation
        Exit Sub
    End If
    If Durum(t) = "DONE" Then
        MsgBox "Tamamlanan is icin hatirlatma taslagi olusturulmaz.", vbInformation
        Exit Sub
    End If
    Set kaynak = KaynakMail(t)
    If kaynak Is Nothing Then
        MsgBox "Orijinal e-posta bulunamadi. Gercek Yanitla/Tumune Yanitla taslagi " & _
               "acilamiyor. 'E-postayi Ac' ile Outlook aramasini deneyin.", vbExclamation
        Exit Sub
    End If

    ' Gercek Outlook yanit nesnesi kullanilir; MessageID / konusma baglami korunur.
    ' Bu metotlar hicbir e-postayi kendiliginden gondermez.
    If tumune Then
        Set taslak = kaynak.ReplyAll
        etiket = "Tumune yanitla"
    Else
        Set taslak = kaynak.Reply
        etiket = "Yanitla"
    End If
    If taslak Is Nothing Then Err.Raise vbObjectError + 127, , "Outlook yanit taslagi olusturamadi."

    ' Display, Outlook imzasini/yazisma alintisini yukledikten sonra metni ekler.
    ' Orijinal HTML govdesi ve e-posta zinciri silinmez.
    taslak.Display
    govde = taslak.HTMLBody
    p = InStr(1, govde, "<body", vbTextCompare)
    If p > 0 Then
        son = InStr(p, govde, ">")
    Else
        son = 0
    End If
    If son > 0 Then
        taslak.HTMLBody = Left$(govde, son) & _
            "<p style='font-family:Calibri,Arial;font-size:11pt'>Merhaba,</p>" & _
            "<p style='font-family:Calibri,Arial;font-size:11pt'>" & _
            "Aşağıdaki yazışmayla ilgili güncel durumu paylaşabilir misiniz?</p>" & _
            "<p style='font-family:Calibri,Arial;font-size:11pt'>Teşekkürler.</p><br>" & _
            Mid$(govde, son + 1)
    Else
        taslak.Body = "Merhaba," & vbCrLf & vbCrLf & _
            "Aşağıdaki yazışmayla ilgili güncel durumu paylaşabilir misiniz?" & _
            vbCrLf & vbCrLf & "Teşekkürler." & vbCrLf & vbCrLf & taslak.Body
    End If

    ' Ozellikle gonderilen e-postadaki Yanitla kendi adresinize donebilir.
    ' Kullanicinin adresleri kontrol etmesi zorunludur; asla otomatik Send yok.
    If Not tumune Then
        MsgBox etiket & " taslagi acildi; GONDERILMEDI." & vbCrLf & _
               "Kaynak e-postayi siz gonderdiyseniz Outlook 'Yanitla' sizi alici yapabilir." & _
               vbCrLf & "Gondermeden once Kime alanini kontrol edin.", vbInformation
    Else
        MsgBox etiket & " taslagi acildi; GONDERILMEDI." & vbCrLf & _
               "Tum alicilari ve varsa CC/Bilgi alanini kontrol edin.", vbInformation
    End If
    Exit Sub
Hata:
    MsgBox "Yanit taslagi hazirlanamadi. Hata " & Err.Number & ": " & _
           Err.Description, vbExclamation, "Is ve Yanit Radari"
End Sub

Private Sub btnTarih_Click()
    On Error GoTo Hata
    Dim t As Outlook.TaskItem, yanit As String, tarih As Date
    Set t = SeciliGorev()
    If t Is Nothing Then
        MsgBox "Önce listeden bir iş seçin.", vbInformation

            t.ReminderSet = False

        Case "1": tarih = DateAdd("h", 1, Now)
        Case "2": tarih = DateAdd("d", 1, Date) + TimeSerial(9, 0, 0)
        Case "3": tarih = DateAdd("d", 3, Date) + TimeSerial(9, 0, 0)

Private Function RadarSayiMi(ByVal yazi As String) As Boolean
    Dim i As Long, kod As Long
    If Len(yazi) = 0 Then Exit Function
    For i = 1 To Len(yazi)
        kod = AscW(Mid$(yazi, i, 1))
        If kod < 48 Or kod > 57 Then Exit Function
    Next i
    RadarSayiMi = True
End Function

Private Function RadarTarihCoz(ByVal yazi As String, ByRef sonuc As Date) As Boolean
    On Error GoTo Bitti
    Dim bolum() As String, tarih() As String, saat() As String
    Dim gun As Long, ay As Long, yil As Long, sa As Long, dk As Long
    Dim gd As Date
    yazi = Replace(Trim$(yazi), "/", ".")
    Do While InStr(yazi, "  ") > 0
        yazi = Replace(yazi, "  ", " ")
    Loop
    bolum = Split(yazi, " ")
    If UBound(bolum) > 1 Then Exit Function
    tarih = Split(bolum(0), ".")
    If UBound(tarih) <> 1 And UBound(tarih) <> 2 Then Exit Function
    If Not RadarSayiMi(tarih(0)) Or Not RadarSayiMi(tarih(1)) Then Exit Function
    gun = CLng(tarih(0))
    ay = CLng(tarih(1))
    yil = Year(Date)
    If UBound(tarih) = 2 Then
        If Len(tarih(2)) <> 4 Or Not RadarSayiMi(tarih(2)) Then Exit Function
        yil = CLng(tarih(2))
    End If
    sa = 9: dk = 0
    If UBound(bolum) = 1 Then
        saat = Split(bolum(1), ":")
        If UBound(saat) <> 1 Then Exit Function
        If Not RadarSayiMi(saat(0)) Or Not RadarSayiMi(saat(1)) Then Exit Function
        sa = CLng(saat(0)): dk = CLng(saat(1))
    End If
    If yil < 2020 Or yil > 2100 Then Exit Function
    If ay < 1 Or ay > 12 Or gun < 1 Or gun > 31 Then Exit Function
    If sa < 0 Or sa > 23 Or dk < 0 Or dk > 59 Then Exit Function
    gd = DateSerial(yil, ay, gun)
    If Year(gd) <> yil Or Month(gd) <> ay Or Day(gd) <> gun Then Exit Function
    sonuc = gd + TimeSerial(sa, dk, 0)
    RadarTarihCoz = (sonuc > Now)
Bitti:
End Function
